    71→// ── helpers ──────────────────────────────────────────────────────────────
    72→function timeAgo(iso: string): string {
    73→  const d = new Date(iso).getTime();
    74→  const diff = Date.now() - d;
    75→  const m = Math.floor(diff / 60000);
    76→  if (m < 1) return "just now";
    77→  if (m < 60) return `${m}m ago`;
    78→  const h = Math.floor(m / 60);
    79→  if (h < 24) return `${h}h ago`;
    80→  const days = Math.floor(h / 24);
    81→  return `${days}d ago`;
    82→}
    83→
    84→function formatViews(v: number): string {
    85→  if (v >= 1_000_000) return (v / 1_000_000).toFixed(1) + "M";
    86→  if (v >= 1_000) return Math.round(v / 1_000) + "K";
    87→  return String(v);
    88→}
    89→
    90→function formatMoney(v: number): string {
    91→  if (v >= 1000) return "$" + (v / 1000).toFixed(1) + "K";
    92→  return "$" + Math.round(v).toLocaleString();
    93→}
    94→
    95→function formatDuration(ms: number): string {
    96→  return (Math.round(ms / 100) / 10).toFixed(1) + "s";
    97→}
    98→
    99→function metricColor(v: number): string {
   100→  if (v >= 80) return "text-emerald-400";
   101→  if (v >= 65) return "text-amber-400";
   102→  return "text-rose-400";
   103→}

   105→// ── Stage stepper ────────────────────────────────────────────────────────
   106→function StageStepper({ currentStage }: { currentStage: string }) {
   107→  const currentIdx = STAGE_INDEX[currentStage] ?? 0;
   108→  return (
   109→    <div className="flex flex-wrap items-start gap-x-1 gap-y-3 rounded-xl border border-border/60 bg-card/30 p-3">
   110→      {PIPELINE.map((stage, idx) => {
   111→        const isPast = idx < currentIdx;
   112→        const isCurrent = idx === currentIdx;
   113→        const isFuture = idx > currentIdx;
   114→        return (
   115→          <div key={stage.key} className="flex items-start">
   116→            <div className="flex w-14 flex-col items-center text-center sm:w-16">
   117→              <div
   118→                className={cn(

   120→                  isPast && "border-emerald-500/40 bg-emerald-500/20 text-emerald-300",
   121→                  isCurrent && "border-emerald-400 bg-emerald-500 text-emerald-950 shadow-lg shadow-emerald-500/30",
   122→                  isFuture && "border-border/60 bg-background/40 text-muted-foreground",

   133→                  "mt-1.5 block text-[9px] leading-tight sm:text-[10px]",
   134→                  isCurrent ? "font-medium text-foreground" : "text-muted-foreground",

   136→              >
   137→                {stage.label}
   138→              </span>

   141→              <div

   155→// ── Approval card ────────────────────────────────────────────────────────

   158→  const [showRevision, setShowRevision] = useState(false);
   159→  const [feedback, setFeedback] = useState("");
   160→  const [pending, setPending] = useState<"approve" | "revise" | "reject" | null>(null);
   161→  const agent = AGENT_MAP[gate.agentType];
   162→
   163→  const decide = useMutation({
   164→    mutationFn: (vars: { decision: "approved" | "rejected" | "revised"; feedback?: string }) =>
   165→      api.decideApproval(gate.id, vars.decision, vars.feedback),
   166→    onMutate: (vars) => {
   167→      setPending(vars.decision === "approved" ? "approve" : vars.decision === "rejected" ? "reject" : "revise");
   168→    },
   169→    onSuccess: (_data, vars) => {
   170→      const msg =
   171→        vars.decision === "approved"
   172→          ? "Approved · advancing pipeline"
   173→          : vars.decision === "rejected"
   174→            ? "Gate rejected"
   175→            : "Revision requested";
   176→      toast.success(msg);
   177→      qc.invalidateQueries({ queryKey: ["project", projectId] });
   178→      qc.invalidateQueries({ queryKey: ["approvals"] });
   179→      qc.invalidateQueries({ queryKey: ["dashboard"] });
   180→      setShowRevision(false);
   181→      setFeedback("");
   182→      setPending(null);
   183→    },
   184→    onError: (e: Error) => {
   185→      toast.error(e.message || "Decision failed");
   186→      setPending(null);
   187→    },
   188→  });
   189→
   190→  return (
   191→    <motion.div
   192→      initial={{ opacity: 0, y: 8 }}
   193→      animate={{ opacity: 1, y: 0 }}
   194→      transition={{ duration: 0.3 }}

   203→              <Lock className="mr-1 h-2.5 w-2.5" /> Approval · {gate.stage}
   204→            </Badge>
   205→            <StatusBadge status={gate.status} />
   206→          </div>
   207→        </div>
   208→        <div className="space-y-4 p-5">

   210→            <h3 className="text-base font-semibold leading-tight">{gate.payload.title}</h3>

   213→            )}

   215→
   216→          {gate.payload.highlights?.length > 0 && (
   217→            <div>
   218→              <p className="mb-1.5 text-[10px] font-medium uppercase tracking-wider text-muted-foreground">

   473→// ── Artifact: Interview panel ────────────────────────────────────────────
   474→function InterviewPanel({ project }: { project: ProjectDetail }) {
   475→  const qc = useQueryClient();
   476→  const [question, setQuestion] = useState("");
   477→  const [answer, setAnswer] = useState("");
   478→  const [selectedQid, setSelectedQid] = useState<string>("");
   479→  const [themeTag, setThemeTag] = useState("");
   480→
   481→  const pending = project.interviews.filter((i) => !i.answer);
   482→  const answered = project.interviews.filter((i) => i.answer);
   483→
   484→  const submitMut = useMutation({
   485→    mutationFn: () => {
   486→      // If a pending question is selected, use its text but submit as a new entry (api takes question + answer)
   487→      const q = selectedQid
   488→        ? project.interviews.find((i) => i.id === selectedQid)?.question ?? ""
   489→        : question.trim();
   490→      return api.submitInterview({
   491→        projectId: project.id,
   492→        question: q,
   493→        answer: answer.trim(),
   494→        themeTag: themeTag.trim() || undefined,
   495→      });
   496→    },
   497→    onSuccess: () => {
   498→      toast.success("Interview answer recorded");
   499→      qc.invalidateQueries({ queryKey: ["project", project.id] });
   500→      setQuestion("");
   501→      setAnswer("");
   502→      setSelectedQid("");
   503→      setThemeTag("");
   504→    },
   505→    onError: (e: Error) => toast.error(e.message || "Failed to submit"),
   506→  });
   507→
   508→  const genQMut = useMutation({
   509→    mutationFn: () =>
   510→      api.runAgent({ agentType: "chief_director", projectId: project.id, input: { action: "interview_questions" } }),
   511→    onSuccess: () => {
   512→      toast.success("Maestro generated new interview questions");
   513→      qc.invalidateQueries({ queryKey: ["project", project.id] });
   514→    },
   515→    onError: (e: Error) => toast.error(e.message || "Failed to generate questions"),
   516→  });
   517→
   518→  return (
   519→    <div className="space-y-4">
   520→      <div className="flex items-center justify-between">
   521→        <p className="text-xs text-muted-foreground">
   522→          {answered.length} answered · {pending.length} pending
   523→        </p>
   524→        <Button

  1182→function ProjectWorkspace({ projectId }: { projectId: string }) {
  1183→  const qc = useQueryClient();
  1184→  const { data: project, isLoading, isError, error } = useQuery({
  1185→    queryKey: ["project", projectId],
  1186→    queryFn: () => api.getProject(projectId),
  1187→    refetchInterval: 8000,
  1188→  });
  1189→
  1190→  const currentStage = project?.stage ?? "opportunity";
  1191→  const currentStageInfo = PIPELINE.find((s) => s.key === currentStage) ?? PIPELINE[0];
  1192→  const currentAgent = AGENT_MAP[currentStageInfo.agent];
  1193→
  1194→  // Find a pending approval gate at the current stage (or any earlier stage if blocked)
  1195→  const pendingGate = useMemo(() => {
  1196→    if (!project) return null;
  1197→    const currentIdx = STAGE_INDEX[currentStage] ?? 0;
  1198→    const gates = project.approvals
  1199→      .filter((g) => g.status === "pending")
  1200→      .sort((a, b) => {
  1201→        const ai = STAGE_INDEX[a.stage] ?? 0;
  1202→        const bi = STAGE_INDEX[b.stage] ?? 0;
  1203→        // Prefer gate at current stage; otherwise the closest earlier pending gate
  1204→        if (ai === currentIdx) return -1;
  1205→        if (bi === currentIdx) return 1;
  1206→        return bi - ai;
  1207→      });
  1208→    return gates[0] ?? null;
  1209→  }, [project, currentStage]);
  1210→
  1211→  const advanceMut = useMutation({
  1212→    mutationFn: () =>
  1213→      api.runAgent({ agentType: "chief_director", projectId, input: { action: "advance" } }),
  1214→    onSuccess: () => {
  1215→      toast.success("Maestro dispatched the next agent");
  1216→      qc.invalidateQueries({ queryKey: ["project", projectId] });
  1217→      qc.invalidateQueries({ queryKey: ["dashboard"] });
  1218→    },
  1219→    onError: (e: Error) => {
  1220→      toast.error(e.message || "Failed to advance pipeline");
  1221→    },
  1222→  });
  1223→
  1224→  if (isLoading) {
  1225→    return (
  1226→      <div className="mx-auto max-w-7xl space-y-4">
  1227→        <div className="h-24 animate-pulse rounded-xl bg-card/40" />

  1229→        <div className="grid gap-4 lg:grid-cols-12">
  1230→          <div className="h-96 animate-pulse rounded-xl bg-card/40 lg:col-span-8" />
  1231→          <div className="h-96 animate-pulse rounded-xl bg-card/40 lg:col-span-4" />
  1232→        </div>
  1233→      </div>
  1234→    );
  1235→  }
  1236→
  1237→  if (isError || !project) {
  1238→    return (
  1239→      <div className="mx-auto max-w-3xl">
  1240→        <Card className="border-rose-500/30 bg-rose-500/5 p-6 text-center text-sm text-rose-300">
  1241→          Failed to load project: {(error as Error)?.message}
  1242→        </Card>
  1243→      </div>
  1244→    );
  1245→  }
  1246→
  1247→  return (
  1248→    <div className="mx-auto max-w-7xl space-y-5">
  1249→      {/* Project header */}
  1250→      <motion.div
  1251→        initial={{ opacity: 0, y: 8 }}
  1252→        animate={{ opacity: 1, y: 0 }}
  1253→        transition={{ duration: 0.3 }}
  1254→      >
  1255→        <Card className="border-border/60 bg-card/40 p-5">
  1256→          <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
  1257→            <div className="min-w-0 flex-1">
  1258→              <div className="flex items-center gap-2 text-xs text-muted-foreground">
  1259→                <button
  1260→                  onClick={() => useApp.setState({ activeProjectId: null })}

  1262→                >
  1263→                  <ArrowLeft className="h-3.5 w-3.5" /> Back to projects
  1264→                </button>
  1265→              </div>
  1266→              <h1 className="mt-1 text-xl font-semibold tracking-tight sm:text-2xl">{project.title}</h1>
  1267→              <div className="mt-1.5 flex flex-wrap items-center gap-2">
  1268→                <Badge variant="outline" className="border-border/60 text-[10px] capitalize text-muted-foreground">
  1269→                  {project.niche}
  1270→                </Badge>
  1271→                <StatusBadge status={project.status} />

  1273→                  {currentStageInfo.label}
